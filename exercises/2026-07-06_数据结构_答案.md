### 第1题 -- 顺序表合并

```c
int seplist_merge(sp dest, sp src)
{
    // 检查 src 是否为空
    if (src->last == -1)
        return 0;
    
    // 检查合并后是否超出容量
    if (dest->last + 1 + src->last + 1 > N)
        return -1;
    
    // 从 dest 的末尾开始，逐个复制 src 的元素
    for (int i = 0; i <= src->last; i++)
    {
        dest->data[dest->last + 1 + i] = src->data[i];
    }
    
    // 更新 dest 的 last
    dest->last += src->last + 1;
    
    return 0;
}
```

### 第2题 -- 单向链表删除中间节点（快慢指针法）

```c
int delete_middle(node_p head)
{
    // 判断是否为空链表或只有一个有效节点
    if (head->next == NULL || head->next->next == NULL)
        return -1;
    
    node_p slow = head;     // 慢指针，指向头节点
    node_p fast = head;     // 快指针，从头节点开始
    
    // 快指针每次走两步，慢指针每次走一步
    // 当快指针走到末尾时，慢指针指向中间节点的前一个节点
    while (fast->next != NULL && fast->next->next != NULL)
    {
        fast = fast->next->next;
        slow = slow->next;
    }
    
    // slow->next 就是要删除的中间节点
    node_p del = slow->next;
    slow->next = del->next;
    free(del);              // 释放内存
    
    return 0;
}
```

### 第3题 -- 括号匹配（栈的正确用法）

```c
int is_balanced(char *expr)
{
    sp stack = create_stack();
    
    for (int i = 0; expr[i] != '\0'; i++)
    {
        char c = expr[i];
        
        // 遇到左括号，入栈
        if (c == '(' || c == '[' || c == '{')
        {
            push_stack(stack, c);
        }
        // 遇到右括号
        else if (c == ')' || c == ']' || c == '}')
        {
            // 如果栈为空，说明右括号没有对应的左括号
            if (is_empty_stack(stack))
            {
                destroy_stack(stack);
                return 0;
            }
            
            char top;
            pop_stack(stack, &top);
            
            // 检查是否匹配
            if ((c == ')' && top != '(') ||
                (c == ']' && top != '[') ||
                (c == '}' && top != '{'))
            {
                destroy_stack(stack);
                return 0;
            }
        }
    }
    
    // 遍历完毕后，栈中应该没有剩余的左括号
    int result = is_empty_stack(stack) ? 1 : 0;
    destroy_stack(stack);
    return result;
}
```

核心思路：
- 遇到左括号 `{ [ (` 就 **入栈**
- 遇到右括号 `} ] )` 就 **出栈** 对比类型是否匹配
- 如果栈空了但还有右括号，说明不匹配
- 如果遍历完了栈还没空，说明有左括号没闭合

### 第4题 -- 循环队列逆序

```c
void reverse_queue(lq_p q)
{
    sp s = create_stack();
    int x;
    
    // 第一步：队列 -> 栈（队列 FIFO 出队，栈 LIFO 入栈）
    while (!is_empty_queue(q))
    {
        dequeue(q, &x);         // 从队头取出元素
        push_stack(s, x);       // 压入栈
    }
    
    // 第二步：栈 -> 队列（栈 LIFO 出栈，队列 FIFO 入队，顺序反转）
    while (!is_empty_stack(s))
    {
        pop_stack(s, &x);       // 从栈顶取出元素
        enqueue(q, x);          // 入队到队尾
    }
}
```

原理图解：
```
原队列顺序: [1, 2, 3, 4, 5]  (从队头到队尾)
出队入栈:   1->2->3->4->5 入栈 (栈底到栈顶: 1,2,3,4,5)
出栈入队:   5->4->3->2->1 入队 (从队头到队尾: 5,4,3,2,1)
最终队列顺序反转
```

### 第5题 -- 判断满二叉树

满二叉树的判定条件：
1. 深度为 k 的满二叉树，节点总数为 **2^k - 1**（根为第0层）
2. 所有节点的度要么是 0（叶子），要么是 2（内部节点），没有度为 1 的节点

方法一（利用节点数公式，代码精简）：

```c
// 计算二叉树深度
int max_depth(bin_node_p root)
{
    if (root == NULL) return 0;
    int left = max_depth(root->lchild);
    int right = max_depth(root->rchild);
    return (left > right ? left : right) + 1;
}

// 计算二叉树节点个数
int count_nodes(bin_node_p root)
{
    if (root == NULL) return 0;
    return count_nodes(root->lchild) + count_nodes(root->rchild) + 1;
}

int is_full_binary_tree(bin_node_p root)
{
    if (root == NULL)
        return 1;  // 空树也是满二叉树
    
    int depth = max_depth(root);      // 树的高度（从1开始计数）
    int nodes = count_nodes(root);    // 节点总数
    
    // 满二叉树节点数 = 2^depth - 1
    int expected = (1 << depth) - 1;  // 2^depth - 1
    
    return nodes == expected;
}
```

方法二（同时检查度为1的节点 + 所有叶子在同一层，更严谨，适合扩展）：

```c
// 检查是否有度为1的节点，同时记录叶子节点深度
int check_full(bin_node_p root, int depth, int *leaf_depth)
{
    if (root == NULL)
        return 1;
    
    // 如果是叶子节点（左右孩子都为空）
    if (root->lchild == NULL && root->rchild == NULL)
    {
        // 如果是第一次遇到叶子节点，记录深度
        if (*leaf_depth == -1)
            *leaf_depth = depth;
        // 如果当前叶子节点深度与之前记录的不同，返回0
        return depth == *leaf_depth;
    }
    
    // 如果有一个孩子为空、另一个不为空，说明度为1，不是满二叉树
    if (root->lchild == NULL || root->rchild == NULL)
        return 0;
    
    // 递归检查左右子树
    return check_full(root->lchild, depth + 1, leaf_depth) &&
           check_full(root->rchild, depth + 1, leaf_depth);
}

int is_full_binary_tree(bin_node_p root)
{
    if (root == NULL)
        return 1;
    
    int leaf_depth = -1;
    return check_full(root, 0, &leaf_depth);
}
