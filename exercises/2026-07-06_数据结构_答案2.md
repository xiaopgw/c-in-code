- **第11题**：选B。单向链表中不知道前驱时，可以通过"数据覆写"法变相删除——把p->next的数据复制到p，然后删除p->next。这是经典面试题。

### 第1题 -- 约瑟夫环（单向循环链表）

```c
int josephus(int n, int k, int m)
{
    int i;
    link_node_p pdel = NULL;  // 指向要删除的节点
    link_node_p ptail = NULL; // 尾指针
    link_node_p pnew = NULL;  // 新节点
    link_node_p h = NULL;     // 头指针
    
    // 1. 创建第一个节点（编号1）
    h = (link_node_p)malloc(sizeof(link_node_t));
    h->data = 1;
    h->next = NULL;
    ptail = h;
    
    // 2. 创建剩余 n-1 个节点，尾插法
    for (i = 2; i <= n; i++)
    {
        pnew = (link_node_p)malloc(sizeof(link_node_t));
        pnew->data = i;
        pnew->next = NULL;
        ptail->next = pnew;
        ptail = pnew;
    }
    // 3. 形成单向循环链表
    ptail->next = h;
    
    // 4. 将头指针移动到开始报数的位置（编号k）
    for (i = 1; i < k; i++)
        h = h->next;
    
    // 5. 循环杀猴子，直到只剩一个节点
    while (h->next != h) // 条件：只剩一个节点时停止
    {
        // 移动到要删除节点的前一个节点（数 m-1 步）
        for (i = 1; i < m - 1; i++)
            h = h->next;
        
        // 删除节点
        pdel = h->next;
        h->next = pdel->next;
        free(pdel);
        
        // 从下一个节点继续报数
        h = h->next;
    }
    
    // 6. 返回最后剩下的节点编号
    int king = h->data;
    free(h);
    return king;
}
```

### 第2题 -- 二叉树层次遍历（使用队列）

```c
void level_order(bin_node_p root)
{
    if (root == NULL)
        return;
    
    // 1. 创建队列
    queue_p q = create_queue();
    
    // 2. 根节点入队
    enqueue(q, root);
    
    // 3. 循环：出队 -> 访问 -> 左右孩子入队
    while (!is_empty_queue(q))
    {
        bin_node_p p;
        dequeue(q, &p);
        
        printf("%d ", p->data);  // 访问当前节点
        
        if (p->lchild != NULL)
            enqueue(q, p->lchild);  // 左孩子入队
        if (p->rchild != NULL)
            enqueue(q, p->rchild);  // 右孩子入队
    }
    
    printf("\n");
}
```

层次遍历执行过程图解（以 `[1] - [2],[3] - [4],[5],null,[6]` 为例）：
```
初始：队列 = [1]
出队1，访问1，入队2,3： 队列 = [2,3]
出队2，访问2，入队4,5： 队列 = [3,4,5]
出队3，访问3，入队6：   队列 = [4,5,6]
出队4，访问4：          队列 = [5,6]
出队5，访问5：          队列 = [6]
出队6，访问6：          队列 = []
输出：1 2 3 4 5 6
```

### 第3题 -- 根据前序和中序还原二叉树

```c
// 辅助递归函数
bin_node_p build(int pre[], int pre_start, int in[], int in_start, int n)
{
    if (n <= 0)
        return NULL;
    
    // 1. 前序的第一个元素是根节点
    int root_data = pre[pre_start];
    
    // 2. 在中序中找到根节点的位置
    int root_pos = 0;
    for (int i = 0; i < n; i++)
    {
        if (in[in_start + i] == root_data)
        {
            root_pos = i;
            break;
        }
    }
    
    // 3. 创建根节点
    bin_node_p root = (bin_node_p)malloc(sizeof(bin_node_t));
    root->data = root_data;
    
    // 4. 左子树节点个数 = root_pos，右子树节点个数 = n - root_pos - 1
    int left_n = root_pos;
    int right_n = n - root_pos - 1;
    
    // 5. 递归构建左右子树
    root->lchild = build(pre, pre_start + 1, in, in_start, left_n);
    root->rchild = build(pre, pre_start + 1 + left_n, in, in_start + root_pos + 1, right_n);
    
    return root;
}

bin_node_p build_tree(int pre[], int in[], int n)
{
    return build(pre, 0, in, 0, n);
}
```

原理说明：
```
前序：[A] B D E C F        根 = A
中序： D B E [A] F C       左子树中序：D B E (3个)，右子树中序：F C (2个)

左子树：               右子树：
前序：B D E             前序：C F
中序：D B E             中序：F C
根 = B                  根 = C
左=D，右=E              左=F
```

### 第4题 -- 链式队列实现

```c
// (1) 创建空链式队列
linkqueue_p create_queue()
{
    // 1. 开辟队列结构体空间
    linkqueue_p q = (linkqueue_p)malloc(sizeof(linkqueue_t));
    if (q == NULL) return NULL;
    
    // 2. 创建头节点
    q->front = q->rear = (link_node_p)malloc(sizeof(link_node_t));
    if (q->front == NULL) return NULL;
    
    // 3. 初始化
    q->front->next = NULL;
    q->len = 0;
    
    return q;
}

// (2) 入队（尾插法）
int enqueue(linkqueue_p q, datatype data)
{
    // 1. 创建新节点
    link_node_p pnew = (link_node_p)malloc(sizeof(link_node_t));
    if (pnew == NULL) return -1;
    pnew->data = data;
    pnew->next = NULL;
    
    // 2. 尾插：新节点链接到尾节点后面
    q->rear->next = pnew;
    
    // 3. 更新尾指针
    q->rear = pnew;
    
    // 4. 长度加一
    q->len++;
    return 0;
}

// (3) 出队（头删法）
datatype dequeue(linkqueue_p q)
{
    // 1. 判空
    if (is_empty(q))
    {
        printf("queue is empty\n");
        return -1;
    }
    
    // 2. pdel 指向要删除的节点（头节点的下一个）
    link_node_p pdel = q->front->next;
    
    // 3. 保存要出队的数据
    datatype temp = pdel->data;
    
    // 4. 头节点跨过 pdel
    q->front->next = pdel->next;
    
    // 5. 如果删除的是最后一个节点，更新 rear
    if (pdel == q->rear)
        q->rear = q->front;
    
    // 6. 释放节点，长度减一
    free(pdel);
    q->len--;
    
    return temp;
}

// (4) 判空
int is_empty(linkqueue_p q)
{
    return q->front == q->rear;
    // 或者：return q->len == 0;
}
```

**链式队列结构示意图：**
```
空队列：   front/rear -> [ 头节点 | NULL ]
非空队列： front -> [头节点] -> [节点1] -> [节点2] -> ... -> [节点n] <- rear
```

### 第5题 -- 二叉树的度统计

```c
void count_degree(bin_node_p root, int *n0, int *n1, int *n2)
{
    if (root == NULL)
        return;
    
    // 判断当前节点的度
    if (root->lchild == NULL && root->rchild == NULL)
    {
        // 度为 0：叶子节点
        (*n0)++;
    }
    else if (root->lchild != NULL && root->rchild != NULL)
    {
        // 度为 2：左右孩子都不为空
        (*n2)++;
    }
    else
    {
        // 度为 1：只有一个孩子
        (*n1)++;
    }
    
    // 递归统计左右子树
    count_degree(root->lchild, n0, n1, n2);
    count_degree(root->rchild, n0, n1, n2);
}
```

使用示例：
```c
int main()
{
    // 构建二叉树
    //          [1]
    //         /   \
    //       [2]   [3]
    //      /  \
    //    [4]  [5]
    
    bin_node_p root = ...;
    int n0 = 0, n1 = 0, n2 = 0;
    count_degree(root, &n0, &n1, &n2);
    printf("n0=%d, n1=%d, n2=%d\n", n0, n1, n2);
    // 输出：n0=2, n1=1, n2=2
    return 0;
}
